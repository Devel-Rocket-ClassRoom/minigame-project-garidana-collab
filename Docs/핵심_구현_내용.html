ScriptableObject Linked-List 구조

QuestData는 ScriptableObject로 만들어져 Unity 에디터에서 에셋 파일로 관리됩니다. PrerequisiteQuest와 NextQuest 필드로 퀘스트 에셋끼리 직접 참조하는 Linked-List 구조를 형성합니다. 문자열 ID 매핑이 필요 없고, 인스펙터에서 에셋을 드래그하면 체인이 완성됩니다.

QuestData.cs — ScriptableObject 에셋 정의
[CreateAssetMenu(fileName = "QuestData", menuName = "Scriptable Objects/QuestData")]
public class QuestData : ScriptableObject
{
    // 목표 정의
    [SerializeField] private QuestObjectiveType objectiveType; // KillMonster / CollectItem / Interact
    [SerializeField] private string targetId;
    [SerializeField] private int requiredAmount = 1;

    // ★ Linked-List 체인 — 에셋 직접 참조
    [SerializeField] private QuestData prerequisiteQuest;  // 선행 퀘스트
    [SerializeField] private QuestData nextQuest;           // 다음 퀘스트

    // 보상
    [SerializeField] private QuestReward reward;          // Gold · EXP · Level
}

QuestManager.cs — GetQuestState()
public QuestState GetQuestState(QuestData quest)
{
    if (IsCompleted(quest))               return QuestState.Completed;
    if (currentQuest == quest)
        return IsCurrentQuestComplete() ? QuestState.ReadyToComplete
                                         : QuestState.InProgress;
    // 선행 미완료이거나 다른 퀘스트 진행 중 → Locked
    if (!IsUnlocked(quest) || currentQuest != null)
        return QuestState.Locked;
    return QuestState.Available;
}

private bool IsUnlocked(QuestData quest)
{
    QuestData prerequisite = quest.PrerequisiteQuest;
    // 선행이 없거나 완료됐으면 해금
    return prerequisite == null || IsCompleted(prerequisite);
}

currentQuest != null 조건이 동시에 하나의 퀘스트만 진행되도록 보장합니다.

QuestManager.cs — ReportProgress()
// 각 시스템에서 타입별로 호출
public void ReportKill(string monsterId)      { ReportProgress(QuestObjectiveType.KillMonster, monsterId, 1); }
public void ReportItemCollected(string itemId) { ReportProgress(QuestObjectiveType.CollectItem, itemId, amount); }
public void ReportInteraction(string targetId) { ReportProgress(QuestObjectiveType.Interact, targetId, 1); }

private void ReportProgress(QuestObjectiveType type, string targetId, int amount)
{
    // 현재 퀘스트 목표 타입과 TargetId 모두 일치할 때만 카운트
    if (!IsCurrentQuestTarget(type, targetId) || amount <= 0) return;
    if (IsCurrentQuestComplete()) return;

    currentAmount = Mathf.Min(currentAmount + amount, currentQuest.RequiredAmount);
    QuestProgressChanged?.Invoke(currentQuest, currentAmount, currentQuest.RequiredAmount);

    if (IsCurrentQuestComplete())
        QuestReadyToComplete?.Invoke(currentQuest);   // UI에 "완료" 버튼 표시
}

이벤트(QuestAccepted, QuestProgressChanged, QuestReadyToComplete, QuestCompleted)로 UI와 완전히 디커플링됩니다. 아이템 드롭도 IsCurrentQuestTarget()으로 현재 목표일 때만 드롭되도록 제어합니다.

public void RefreshStage(int level)
{
    AttackStageData selectedStage = null;
    for (int i = 0; i < _attackStages.Length; i++)
    {
        AttackStageData stage = _attackStages[i];
        if (stage != null && level >= stage.UnlockLevel)
            selectedStage = stage;  // 계속 덮어써서 마지막 것(최고)을 선택
    }
    _currentStage = selectedStage;
}

// ① PBKDF2: Salt에서 암호화 키 + HMAC 키를 동시에 유도 (12만 회 반복)
using (Rfc2898DeriveBytes deriveBytes = new Rfc2898DeriveBytes(
    password, salt, 120000, HashAlgorithmName.SHA256))
{
    encryptionKey = deriveBytes.GetBytes(32);  // AES-256 키
    hmacKey       = deriveBytes.GetBytes(32);  // HMAC 키
}

// ② AES-256-CBC: 매 저장마다 새 IV 생성 → 재사용 공격 차단
aes.Mode = CipherMode.CBC;
aes.Padding = PaddingMode.PKCS7;
aes.GenerateIV();   // 랜덤 16바이트, 매번 새로 생성

// ③ HMAC-SHA256: format+version+salt+iv+cipherText 전체 서명
//    로드 시 서명 불일치 → 변조 파일로 판단하고 거부
byte[] authenticatedData = BuildAuthenticatedData(version, iterations, salt, iv, cipherText);
byte[] hmac = ComputeHmac(hmacKey, authenticatedData);

// ④ 타이밍 어택 방지: 고정 시간 비교
private static bool FixedTimeEquals(byte[] left, byte[] right)
{
    int difference = 0;
    for (int i = 0; i < left.Length; i++)
        difference |= left[i] ^ right[i];  // 일찍 return하지 않아 시간이 일정함
    return difference == 0;
}

private void SwapEquip(ref ItemData slot, ItemData newItem, Transform mountParent)
{
    if (slot != null) {
        RemoveStats(slot);                              // 이전 장비 스탯 제거
        SetModelActive(slot.equipObjectName, mountParent, false); // 모델 숨기기
    }
    slot = newItem;
    ApplyStats(slot);                                    // 새 장비 스탯 적용
    SetModelActive(slot.equipObjectName, mountParent, true);  // 모델 보이기
    EquipmentChanged?.Invoke();
}

private void SetModelActive(string equipName, Transform mountParent, bool active)
{
    // mountParent 직접 → 재귀 탐색 → 전체 재귀 순서로 폴백
    Transform target = mountParent.Find(equipName)
                    ?? FindChildRecursive(mountParent, equipName)
                    ?? FindChildRecursive(transform, equipName);
    target.gameObject.SetActive(active);
}

{
    float desiredX = localPoint.x + TooltipOffsetX;
    float desiredY = localPoint.y + TooltipOffsetY;

    if (desiredX + width > rightBound)
        desiredX = localPoint.x - TooltipOffsetX - width;   // 왼쪽으로 반전
    if (desiredY + height > topBound)
        desiredY = localPoint.y - TooltipOffsetY - height;  // 아래로 반전

    // 그래도 벗어나면 ScreenPadding으로 클램프
    desiredX = Mathf.Clamp(desiredX, leftBound, rightBound - width);
    desiredY = Mathf.Clamp(desiredY, bottomBound, topBound - height);
}

공통 패턴: 우선순위 기반 ActiveZones 리스트

라이팅(LightingZoneTrigger)과 사운드(BgmZoneTrigger) 모두 동일한 패턴을 사용합니다. 트리거 진입·이탈 시마다 전체 ActiveZones 리스트를 재평가해 priority가 가장 높은 존을 적용합니다. 구역이 겹쳐도 항상 올바른 존이 우선 적용됩니다.

LightingZoneTrigger.cs — 진입/이탈 + 우선순위 평가
private static readonly List<LightingZoneTrigger> ActiveZones = new();

private void OnTriggerEnter(Collider other)
{
    if (!other.CompareTag("Player")) return;
    if (!ActiveZones.Contains(this)) ActiveZones.Add(this);  // 진입한 존 추가
    ApplyHighestPriorityZone();
}
private void OnTriggerExit(Collider other)
{
    ActiveZones.Remove(this);    // 나간 존 제거
    ApplyHighestPriorityZone();  // 재평가
}

private static void ApplyHighestPriorityZone()
{
    LightingZoneTrigger bestZone = null;
    for (int i = 0; i < ActiveZones.Count; i++)
        if (bestZone == null || ActiveZones[i].priority > bestZone.priority)
            bestZone = ActiveZones[i];  // priority 최고값 선택

    LightingZoneController.Instance?.BlendTo(bestZone.profile, bestZone.zoneVolume, bestZone.fadeDuration);
}

라이팅 블렌딩 코루틴 — SmoothStep 이징

LightingZoneController.cs — BlendRoutine()
private IEnumerator BlendRoutine(LightingZoneProfile profile, Volume targetVolume, float duration)
{
    // 시작값 스냅샷
    Color startLightColor   = directionalLight.color;
    float startLightIntensity = directionalLight.intensity;
    Color startAmbient      = RenderSettings.ambientLight;
    Color startFogColor     = RenderSettings.fogColor;

    float elapsed = 0f;
    while (elapsed < duration)
    {
        elapsed += Time.deltaTime;
        // SmoothStep → 선형이 아닌 ease-in/out 이징 적용
        float t = Mathf.SmoothStep(0f, 1f, Mathf.Clamp01(elapsed / duration));

        directionalLight.color     = Color.Lerp(startLightColor, profile.directionalColor, t);
        directionalLight.intensity = Mathf.Lerp(startLightIntensity, profile.directionalIntensity, t);
        RenderSettings.ambientLight = Color.Lerp(startAmbient, profile.ambientColor, t);
        RenderSettings.fogDensity   = Mathf.Lerp(startFogDensity, targetFogDensity, t);

        // 포스트프로세싱 Volume 교차 페이드: 이전 → 0, 새 것 → 1
        if (previousVolume != null) previousVolume.weight = Mathf.Lerp(startPrevWeight, 0f, t);
        if (targetVolume != null)   targetVolume.weight   = Mathf.Lerp(startTargetWeight, 1f, t);

        yield return null;
    }
}

새 블렌드 요청이 오면 현재 코루틴을 StopCoroutine으로 즉시 중단하고 현재 값에서 다시 시작합니다. 구역을 빠르게 이동해도 팝핑 없이 부드럽게 전환됩니다.

    toSource.clip = clip;
    toSource.volume = 0f;
    toSource.Play();   // 새 BGM 무음으로 시작

    while (elapsed < fadeDuration)
    {
        fromSource.volume = Mathf.Lerp(fromStartVolume, 0f, t);  // 기존 fade out
        toSource.volume   = Mathf.Lerp(0f, 1f, t);              // 새 것 fade in
        yield return null;
    }
    StopBgmSource(fromSource);
    SwapBgmSources(type);   // A↔B 역할 교체 (다음 전환 준비)
}

// 볼륨은 AudioMixer를 통해 dB로 제어 → 마스터/BGM/SFX 독립 조절
private void SetMixerVolume(string paramName, float value)
{
    // 0 이하 방지 후 선형 → dB 변환: 20 * log10(value)
    float db = Mathf.Log10(Mathf.Max(value, 0.0001f)) * 20f;
    _audioMixer.SetFloat(paramName, db);
}

Time.unscaledDeltaTime을 사용하므로 timeScale = 0(일시 정지) 상태에서도 BGM 페이드가 자연스럽게 진행됩니다. 볼륨 설정은 PlayerPrefs에 저장됩니다.
